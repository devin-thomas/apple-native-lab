# What a built product may link and carry, per build profile and platform.
# script/build_manifest.py checks every product it builds against these rules, and fails a
# profile whose product breaks one. A new framework or entitlement is therefore a reviewed line
# here, never an accident. See docs/BUILD_AND_DISTRIBUTION.md.
#
# Platforms: macos, ios, watchos, tvos, visionos (simulator and device builds share a name).
# A profile or platform of * matches any. Text after # is a comment; keep the reason.
#
#   link         <profile> <platform> <Framework>   executables may link /System/Library/Frameworks/<Framework>
#   entitlement  <profile> <platform> <key>         executables may declare this entitlement
#   unsupported  <platform> <Framework>             no executable on the platform may link it, whatever `link` says
#   purpose      <Framework> <Info.plist key>       a Store-lane product that links the framework must declare the key

# Every host
link         *          *        Foundation          # base
link         *          *        CoreFoundation      # base under Foundation: URL resource-value keys (LabStaging and the share inbox, LAB-007)
link         *          *        SwiftUI             # host interface
link         *          ios      UIKit               # linked by SwiftUI apps on iOS
link         *          watchos  UIKit               # linked by SwiftUI apps on watchOS
link         *          tvos     UIKit               # linked by SwiftUI apps on tvOS

# CoreLocal: capability status probes in LabSupport (CORE-004); none of them prompts
link         CoreLocal  macos    AVFoundation        # camera and microphone status
link         CoreLocal  macos    CoreBluetooth       # Bluetooth authorization status
link         CoreLocal  macos    FoundationModels    # on-device language model availability
link         CoreLocal  macos    Security            # reads the running build's own entitlements
link         CoreLocal  macos    Speech              # speech recognition status and on-device assets
link         CoreLocal  macos    CryptoKit           # LabStore demo fixture: SHA-256 of the bundled seed
link         CoreLocal  macos    AppKit              # accessibility announcements SwiftUI does not supply
link         CoreLocal  ios      ARKit               # world tracking and scene reconstruction support
link         CoreLocal  ios      AVFoundation        # camera and microphone status
link         CoreLocal  ios      CoreBluetooth       # Bluetooth authorization status
link         CoreLocal  ios      CryptoKit           # LabStore demo fixture: SHA-256 of the bundled seed
link         CoreLocal  ios      FoundationModels    # on-device language model availability
link         CoreLocal  ios      NearbyInteraction   # Ultra Wideband device capabilities
link         CoreLocal  ios      Speech              # speech recognition status and on-device assets
entitlement  CoreLocal  macos    com.apple.security.app-sandbox   # the Mac host is sandboxed

# CoreLocal: Action Atlas (LAB-001). App Intents need no entitlement; the in-app browser is the fallback
link         CoreLocal  macos    AppIntents          # App Intents, entities, and queries over the operation service
link         CoreLocal  ios      AppIntents          # same, iPhone host
link         CoreLocal  macos    CoreTransferable    # the action browser's Share button for an export (ShareLink)
link         CoreLocal  ios      CoreTransferable    # same, iPhone host

# CoreLocal: Access as a Superpower (LAB-035). The chart's Audio Graph; the summary and table are the fallback
link         CoreLocal  macos    Accessibility       # AXChartDescriptor for the archive chart's Audio Graph (SwiftUI accessibilityChartDescriptor)
link         CoreLocal  ios      Accessibility       # same, iPhone host
# SystemSurfaces: Share Ingress Station (LAB-007). LabPhoneSurfaces is the iPhone host built from
# LabPhone's sources, so it links what LabPhone links, for the same reasons. The share extension it
# embeds links only Foundation, SwiftUI, UIKit, and CryptoKit. Only this profile may carry the App Group.
link         SystemSurfaces ios  ARKit               # host variant: world tracking support, as LabPhone
link         SystemSurfaces ios  AVFoundation        # host variant: camera and microphone status, as LabPhone
link         SystemSurfaces ios  Accessibility       # host variant: the Audio Graph chart descriptor, as LabPhone
link         SystemSurfaces ios  AppIntents          # host variant: Action Atlas intents, as LabPhone; the widget extension: Surface Deck's toggle and launch action
link         SystemSurfaces ios  CoreBluetooth       # host variant: Bluetooth authorization status, as LabPhone
link         SystemSurfaces ios  CoreTransferable    # host variant: the action browser's ShareLink, as LabPhone
link         SystemSurfaces ios  CryptoKit           # host and extensions: SHA-256 of staged content and the demo seed (LabDomain)
link         SystemSurfaces ios  FoundationModels    # host variant: on-device language model availability, as LabPhone
link         SystemSurfaces ios  NearbyInteraction   # host variant: Ultra Wideband capabilities, as LabPhone
link         SystemSurfaces ios  Speech              # host variant: speech recognition status, as LabPhone
entitlement  SystemSurfaces ios  com.apple.security.application-groups   # share inbox folder the extension writes and the host variant reads; a device needs a paid team
# CoreLocal: Portable Objects (LAB-008). CoreFoundation is covered by the base line above.
entitlement  CoreLocal  macos    com.apple.security.files.user-selected.read-write   # Import and Export file dialogs (the declared fallback); the sandbox shows no panel without it

# CoreLocal and SystemSurfaces: Trust Desk (LAB-041). The keychain record is the app's own.
# LocalAuthentication is device-owner confirmation, not an account. AuthenticationServices is
# linked for the platform-passkey symbol; this build does not present a system passkey.
link         CoreLocal      macos    LocalAuthentication     # device-owner confirmation for the sealed record
link         CoreLocal      ios      LocalAuthentication     # same, iPhone host
link         CoreLocal      ios      Security                # the scoped keychain record
link         CoreLocal      macos    AuthenticationServices  # platform passkey provider symbol; the ceremony is a simulation
link         CoreLocal      ios      AuthenticationServices  # same, iPhone host
link         SystemSurfaces ios      LocalAuthentication     # host variant, as LabPhone
link         SystemSurfaces ios      Security                # host variant, as LabPhone
link         SystemSurfaces ios      AuthenticationServices  # host variant, as LabPhone
purpose      LocalAuthentication NSFaceIDUsageDescription    # device-owner confirmation can use Face ID

# SystemSurfaces: Surface Deck (LAB-004). The host variant writes the session snapshot into the same
# App Group and asks WidgetKit to reload only when it changes. The widget and Control extension it
# embeds links only Foundation, SwiftUI, UIKit, AppIntents, _AppIntents_SwiftUI, CryptoKit (through
# LabDomain), and WidgetKit: it reads the snapshot file and never the store or a model. CoreLocal never
# links WidgetKit.
link         SystemSurfaces ios  WidgetKit           # the widget and Controls; the host variant's reloadTimelines and reloadControls
link         SystemSurfaces ios  _AppIntents_SwiftUI # the widget's Toggle(isOn:intent:) comes from SwiftUI's App Intents overlay

# Companions: the Watch host's capability status probes (CORE-004)
link         Companions watchos  AVFAudio            # record permission and audio input
link         Companions watchos  CoreBluetooth       # Bluetooth authorization status
link         Companions watchos  NearbyInteraction   # Ultra Wideband device capabilities

# Companions: the Apple TV host (CORE-013). It links LabSupport and LabCatalog only, and declares no
# entitlement: no `entitlement` line names tvOS, so any entitlement fails the manifest.
link         Companions tvos     CoreBluetooth       # Bluetooth authorization status (CBManager.authorization); no manager is created

# SDK modules that exist but do not provide the lab's API on these platforms (CORE-004 notes)
unsupported  macos      ARKit                        # the macOS SDK's ARKit has no ARWorldTrackingConfiguration
unsupported  macos      NearbyInteraction            # NISession is unavailable on macOS
unsupported  watchos    FoundationModels             # SystemLanguageModel is unavailable on watchOS
unsupported  tvos       FoundationModels             # SystemLanguageModel is unavailable on tvOS
unsupported  tvos       Speech                       # SFSpeechRecognizer is unavailable on tvOS
